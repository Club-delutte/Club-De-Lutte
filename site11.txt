  <script>
    // State management and LocalStorage keys
    const STORAGE_KEY = 'club_lutte_data';

    const defaultData = {
      clubName: 'Club de Lutte',
      heroBadge: 'Saison 2026-2027 — Inscriptions Ouvertes',
      heroTitle: 'Forgez Votre Corps & Votre Esprit',
      heroSubtitle: 'Rejoignez l\'académie de lutte d\'élite. Discipline, force et technique enseignées par des champions pour tous les niveaux.',
      aboutTitle: 'L\'Excellence par la Tradition et l\'Effort',
      aboutText: 'Fondé avec la passion d\'inculquer la maîtrise du corps et la force mentale, notre club accueille tous les passionnés de lutte Libre et Gréco-Romaine. Que vous visiez la compétition ou le perfectionnement personnel, nos installations modernes et notre encadrement de haut niveau garantissent votre progression.',
      aboutImg: '',
      address: '12 Rue du Sport, 75000 Paris',
      phone: '+33 1 23 45 67 89',
      email: 'contact@clublutte.fr'
    };

    function getData() {
      const stored = localStorage.getItem(STORAGE_KEY);
      return stored ? JSON.parse(stored) : defaultData;
    }

    function renderData() {
      const data = getData();
      
      document.getElementById('displayClubName').innerText = data.clubName;
      document.getElementById('footerClubName').innerText = data.clubName;
      document.getElementById('displayHeroBadge').innerText = data.heroBadge;
      document.getElementById('displayHeroTitle').innerHTML = data.heroTitle.includes('span') 
        ? data.heroTitle 
        : data.heroTitle.replace(/(Corps|Esprit)/g, '<span class="accent">$1</span>');
      document.getElementById('displayHeroSubtitle').innerText = data.heroSubtitle;
      
      document.getElementById('displayAboutTitle').innerText = data.aboutTitle;
      document.getElementById('displayAboutText').innerText = data.aboutText;
      
      if (data.aboutImg) {
        document.getElementById('aboutImg').src = data.aboutImg;
        document.getElementById('aboutImg').style.display = 'block';
        document.getElementById('aboutImgPlaceholder').style.display = 'none';
      } else {
        document.getElementById('aboutImg').style.display = 'none';
        document.getElementById('aboutImgPlaceholder').style.display = 'flex';
      }

      document.getElementById('displayAddress').innerText = data.address;
      document.getElementById('displayPhone').innerText = data.phone;
      document.getElementById('displayEmail').innerText = data.email;
    }

    // Modal Control Functions
    function openLoginModal() {
      document.getElementById('loginModal').classList.add('open');
      document.getElementById('loginPassword').focus();
    }

    function closeLoginModal() {
      document.getElementById('loginModal').classList.remove('open');
      document.getElementById('loginError').style.display = 'none';
      document.getElementById('loginPassword').value = '';
    }

    function checkPassword() {
      const pwd = document.getElementById('loginPassword').value;
      if (pwd === 'admin' || pwd === '1234') {
        closeLoginModal();
        openAdminPanel();
      } else {
        document.getElementById('loginError').style.display = 'block';
      }
    }

    function openAdminPanel() {
      const data = getData();
      
      document.getElementById('inputClubName').value = data.clubName;
      document.getElementById('inputHeroBadge').value = data.heroBadge;
      document.getElementById('inputHeroTitle').value = data.heroTitle;
      document.getElementById('inputHeroSubtitle').value = data.heroSubtitle;
      
      document.getElementById('inputAboutTitle').value = data.aboutTitle;
      document.getElementById('inputAboutText').value = data.aboutText;
      
      if (data.aboutImg) {
        const prev = document.getElementById('aboutImagePreview');
        prev.src = data.aboutImg;
        prev.classList.add('visible');
      }

      document.getElementById('inputAddress').value = data.address;
      document.getElementById('inputPhone').value = data.phone;
      document.getElementById('inputEmail').value = data.email;

      document.getElementById('adminPanel').classList.add('open');
    }

    function closeAdminPanel() {
      document.getElementById('adminPanel').classList.remove('open');
    }

    function handleImageUpload(input, targetKey) {
      if (input.files && input.files[0]) {
        const reader = new FileReader();
        reader.onload = function(e) {
          const prev = document.getElementById(targetKey + 'ImagePreview');
          prev.src = e.target.result;
          prev.classList.add('visible');
          prev.dataset.base64 = e.target.result;
        }
        reader.readAsDataURL(input.files[0]);
      }
    }

    function removeImage(targetKey) {
      const prev = document.getElementById(targetKey + 'ImagePreview');
      prev.src = '';
      prev.classList.remove('visible');
      delete prev.dataset.base64;
    }

    function saveAdminData() {
      const data = getData();
      
      data.clubName = document.getElementById('inputClubName').value;
      data.heroBadge = document.getElementById('inputHeroBadge').value;
      data.heroTitle = document.getElementById('inputHeroTitle').value;
      data.heroSubtitle = document.getElementById('inputHeroSubtitle').value;
      
      data.aboutTitle = document.getElementById('inputAboutTitle').value;
      data.aboutText = document.getElementById('inputAboutText').value;
      
      const prev = document.getElementById('aboutImagePreview');
      if (prev.dataset.base64) {
        data.aboutImg = prev.dataset.base64;
      } else if (!prev.classList.contains('visible')) {
        data.aboutImg = '';
      }

      data.address = document.getElementById('inputAddress').value;
      data.phone = document.getElementById('inputPhone').value;
      data.email = document.getElementById('inputEmail').value;

      localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
      renderData();
      closeAdminPanel();
      showToast('Modifications enregistrées !');
    }

    // Interactive UI Helpers
    function toggleMobileMenu() {
      document.getElementById('mobileMenu').classList.toggle('open');
      document.getElementById('mobileToggle').classList.toggle('active');
    }

    function showToast(msg) {
      const t = document.getElementById('toast');
      t.innerText = '✓ ' + msg;
      t.classList.add('show');
      setTimeout(() => t.classList.remove('show'), 3000);
    }

    function openLightbox(symbol) {
      const lightbox = document.getElementById('lightbox');
      const lightboxImg = document.getElementById('lightboxImg');
      // Create a dynamic placeholder SVG image preview for demo lightbox
      const svgString = `<svg xmlns="http://www.w3.org/2000/svg" width="600" height="400" viewBox="0 0 600 400"><rect width="600" height="400" fill="#171717"/><text x="50%" y="50%" fill="#EF4444" font-size="80" text-anchor="middle" dominant-baseline="central">${symbol}</text></svg>`;
      lightboxImg.src = 'data:image/svg+xml;charset=utf-8,' + encodeURIComponent(svgString);
      lightbox.classList.add('open');
    }

    function closeLightbox() {
      document.getElementById('lightbox').classList.remove('open');
    }

    // Scroll Animations
    window.addEventListener('scroll', () => {
      const navbar = document.getElementById('navbar');
      if (window.scrollY > 50) {
        navbar.classList.add('scrolled');
      } else {
        navbar.classList.remove('scrolled');
      }

      document.querySelectorAll('.fade-in-up').forEach(el => {
        const rect = el.getBoundingClientRect();
        if (rect.top < window.innerHeight - 100) {
          el.classList.add('visible');
        }
      });
    });

    // Initialize on load
    window.addEventListener('DOMContentLoaded', () => {
      renderData();
      // Trigger animations for visible elements on load
      setTimeout(() => {
        document.querySelectorAll('.fade-in-up').forEach(el => {
          const rect = el.getBoundingClientRect();
          if (rect.top < window.innerHeight) {
            el.classList.add('visible');
          }
        });
      }, 100);
    });
  </script>
